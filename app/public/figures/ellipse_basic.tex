\documentclass[border=20pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{calc,arrows.meta,decorations.pathmorphing}
\begin{document}
\begin{tikzpicture}[scale=0.9]
  % Axes
  \draw[->, gray!60, thin] (-5.5,0) -- (5.5,0) node[right] {$x$};
  \draw[->, gray!60, thin] (0,-4.5) -- (0,4.5) node[above] {$y$};

  % Ellipse: x²/16 + y²/9 = 1, a=4, b=3
  \draw[thick, blue!70!black] (0,0) ellipse (4 and 3);

  % Foci: (±√7, 0) ≈ (±2.646, 0)
  \coordinate (F1) at (-2.646, 0);
  \coordinate (F2) at (2.646, 0);
  \coordinate (O)  at (0,0);
  \coordinate (P)  at (4,0);

  % Segments from foci to P
  \draw[thick, blue!50!black, dashed] (F1) -- (P);
  \draw[thick, blue!50!black, dashed] (F2) -- (P);

  % Vertices
  \fill[blue!70!black] (-4,0) circle (2pt) node[below left] {$A_1$};
  \fill[blue!70!black] (4,0)  circle (2pt) node[below right] {$A_2$};
  \fill[blue!70!black] (0,-3) circle (2pt) node[below left] {$B_1$};
  \fill[blue!70!black] (0,3)  circle (2pt) node[above left] {$B_2$};

  % Foci dots
  \fill[red!70!black] (F1) circle (2pt) node[below] {$F_1$};
  \fill[red!70!black] (F2) circle (2pt) node[below] {$F_2$};

  % Center
  \fill[black] (O) circle (2pt) node[below left] {$O$};

  % Point P on ellipse
  \fill[blue] (P) circle (2pt) node[above right] {$P$};

  % Distance labels
  \node[blue!50!black] at (1.2, 0.5) {$|PF_1|$};
  \node[blue!50!black] at (3.3, 0.5) {$|PF_2|$};

  % Note at bottom
  \node[text=gray] at (0, -4.2) {$|PF_1|+|PF_2|=2a=8$};
\end{tikzpicture}
\end{document}
