\begin{document}
\begin{tikzpicture}[scale=0.75]
  % Axes
  \draw[->, gray!60, thin] (-6.5,0) -- (6.5,0) node[right] {$x$};
  \draw[->, gray!60, thin] (0,-4.5) -- (0,4.5) node[above] {$y$};

  % Ellipse: x^2/25 + y^2/9 = 1, a=5, b=3, c=4
  \draw[thick, blue!70!black] (0,0) ellipse (5 and 3);

  % Foci: F1(-4,0), F2(4,0)
  \coordinate (F1) at (-4, 0);
  \coordinate (F2) at (4, 0);
  \coordinate (O)  at (0,0);

  % Point P on ellipse (choose a position where angle looks ~60°)
  % At param angle ~70°: (5cos70, 3sin70) ≈ (1.71, 2.82)
  \coordinate (P) at (1.71, 2.82);

  % Triangle F1-P-F2
  \draw[thick, blue!50!black] (F1) -- (P) -- (F2) -- cycle;

  % Angle at P (60°)
  \draw pic["$60^\circ$", draw=orange!80!black, thick, angle eccentricity=1.4, angle radius=22] {angle=F1--P--F2};

  % Points
  \fill[red!70!black]  (F1) circle (2.5pt) node[below] {$F_1(-4,0)$};
  \fill[red!70!black]  (F2) circle (2.5pt) node[below] {$F_2(4,0)$};
  \fill[blue!70!black] (P)  circle (2.5pt) node[above right] {$P$};
  \fill[black]         (O)  circle (1.5pt) node[below left]  {$O$};

  % Labels
  \node[blue!50!black] at (-1.2, 1.2) {$r_1$};
  \node[blue!50!black] at (3.0, 1.5)  {$r_2$};
  \node[gray] at (0, -0.4) {$2c=8$};

  % Equation note
  \node[text=gray, font=\small] at (0, -4.0) {$\dfrac{x^2}{25}+\dfrac{y^2}{9}=1\qquad S=b^2\tan30^\circ=9\cdot\frac{\sqrt3}{3}=3\sqrt3$};
\end{tikzpicture}
