\documentclass[border=20pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usetikzlibrary{calc,arrows.meta}
\begin{document}
\begin{tikzpicture}[scale=0.85]
  % Axes
  \draw[->, gray!60, thin] (-2.5,0) -- (5.5,0) node[right] {$x$};
  \draw[->, gray!60, thin] (0,-4.5) -- (0,4.5) node[above] {$y$};

  % Parabola y^2 = 4x → y = ±2√x
  \draw[thick, blue!70!black, domain=0:5, samples=100, variable=\x]
    plot (\x, {2*sqrt(\x)});
  \draw[thick, blue!70!black, domain=0:5, samples=100, variable=\x]
    plot (\x, {-2*sqrt(\x)});

  % Focus F(1,0)
  \coordinate (F) at (1, 0);

  % Directrix x = -1
  \draw[gray!50, dashed, thin] (-1, -4.2) -- (-1, 4.2);
  \node[gray!60, font=\footnotesize] at (-1, 4.4) {$x=-1$};

  % Chord AB through F (approximate: y = 2(x-1))
  % Intersection of y=2(x-1) with y^2=4x:
  % 4(x-1)^2 = 4x → (x-1)^2 = x → x^2-2x+1=x → x^2-3x+1=0
  % x = (3±√5)/2 ≈ (3±2.236)/2 → x1≈2.618, x2≈0.382
  % y1 = 2(2.618-1) = 3.236, y2 = 2(0.382-1) = -1.236
  \coordinate (A) at (2.618, 3.236);
  \coordinate (B) at (0.382, -1.236);

  % Chord line
  \draw[thick, blue!50!black] (A) -- (B);
  % Extend line slightly
  \draw[blue!50!black, thin, dashed] (B) -- (-0.3, -2.6);
  \draw[blue!50!black, thin, dashed] (A) -- (4.2, 6.4);

  % Points
  \fill[red!70!black]  (F) circle (2.5pt) node[below left] {$F$};
  \fill[blue!70!black] (A) circle (2.5pt) node[above right] {$A$};
  \fill[blue!70!black] (B) circle (2.5pt) node[below right] {$B$};
\end{tikzpicture}
\end{document}
